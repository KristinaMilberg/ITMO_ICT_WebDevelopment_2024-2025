    • POST: сохраняет данные и перенаправляет на страницу входа.
      class RegisterView(View):
    def get(self, request):
        form = UserRegistrationForm()
        return render(request, 'account/register.html', {'form': form})

    def post(self, request):
        form = UserRegistrationForm(request.POST)
        if form.is_valid():
            form.save()
            return redirect('/login/')
        return render(request, 'account/register.html', {'form': form})

    Бронирование комнаты

    RoomReservationView: отображает и обрабатывает процесс бронирования комнаты.

    • GET: показывает доступные даты.
    • POST: проверяет введённые даты и создаёт бронирование.
      class RoomReservationView(LoginRequiredMixin, DetailView):
    model = Room
    template_name = "reservation/make_reservation.html"
    context_object_name = "room"

    def get_context_data(self, **kwargs):
        context = super().get_context_data(**kwargs)
        room = self.get_object()

        dt_from = parse_or_return(self.request.GET.get('dt_from') or datetime.datetime.today().date())
        dt_till = parse_or_return(self.request.GET.get('dt_till') or dt_from + datetime.timedelta(days=30))

        free_dates = get_free_dates(room, dt_from, dt_till)

        context['free_dates'] = free_dates
        return context

    def post(self, request, *args, **kwargs):
        self.object = self.get_object()
        check_in_date = parse_or_return(request.POST.get('check_in_date'))
        check_out_date = parse_or_return(request.POST.get('check_out_date'))
        room = self.get_object()
        errors = []

        if check_in_date < datetime.datetime.today().date():
            errors.append('The check-in date is in the future')

        if not errors and check_in_date > check_out_date:
            errors.append('The check-in date must be later than the check-out date.')

        free_dates = get_free_dates(room, check_in_date, check_out_date)
        if not errors and len(free_dates) != (check_out_date - check_in_date).days + 1:
            errors.append("The selected dates are not available for this room.")

        if errors:
            context = self.get_context_data()
            context['errors'] = errors
            context['check_in_date'] = check_in_date
            context['check_out_date'] = check_out_date
            return self.render_to_response(context)

        Reservation.objects.create(
            guest=request.user,
            room=room,
            check_in_date=check_in_date,
            check_out_date=check_out_date,
        )

        messages.success(request, "Your reservation has been created and is pending confirmation.")
        return self.get(request, *args, **kwargs)

    return render(request, 'reservation/reservation_list.html', {'reservations': reservations})

    Редактирование бронирования

    edit_reservation: позволяет редактировать существующее бронирование.

    @login_required
def edit_reservation(request, pk):
    reservation = Reservation.objects.get(id=pk)

    if reservation.guest != request.user:
        return redirect('reservation_list')

    if request.method == 'POST':
        form = ReservationForm(request.POST, instance=reservation)
        check_in_date = parse_or_return(form.data['check_in_date'])
        check_out_date = parse_or_return(form.data['check_out_date'])

        if check_in_date < datetime.datetime.today().date():
            form.add_error('check_in_date', 'The check-in date is in the future')

        if not form.errors and check_in_date > check_out_date:
            form.add_error('check_out_date', 'The check-in date must be later than the check-out date.')

        free_dates = get_free_dates(reservation.room, check_in_date, check_out_date, exclude=reservation.id)
        if not form.errors and len(free_dates) != (check_out_date - check_in_date).days + 1:
            form.add_error('check_out_date', "The selected dates are not available for this room.")

        if form.is_valid():
            form.save()
            return redirect('reservation_list')
    else:
        form = ReservationForm(instance=reservation)
    return render(request, 'reservation/edit_reservation.html', {'form': form, 'reservation': reservation})

    Удаление бронирования

    delete_reservation: удаляет бронирование пользователя.

    @login_required
def delete_reservation(request, pk):
    reservation = Reservation.objects.get(id=pk)
    if reservation.guest == request.user:
        reservation.delete()
    return redirect('reservation_list')

    @login_required
def leave_review(request, pk):
    reservation = get_object_or_404(Reservation, pk=pk)

    if reservation.guest != request.user:
        return redirect('reservation_list')
    if request.method == 'POST':
        form = ReviewForm(request.POST)
        if form.is_valid():
            review = form.save(commit=False)
            review.reservation = reservation
            review.save()
            return redirect('hotel_list')
    else:
        form = ReviewForm()
    return render(request, 'reservation/leave_review.html', {'form': form, })

    Обновление отзыва

    ReviewUpdateView: позволяет редактировать существующий отзыв.

    class ReviewUpdateView(LoginRequiredMixin, UpdateView):
    model = Review
    form_class = ReviewForm
    template_name = 'reservation/leave_review.html'
    success_url = reverse_lazy('reservation_list')

    def get_object(self):
        return Review.objects.get(reservation=self.kwargs['pk'])

    def form_valid(self, form):
        if form.instance.reservation.guest == self.request.user:
            return super().form_valid(form)
        raise ValidationError(form.errors)

    Итог по представлениям

    Управление аккаунтом:

    • Регистрация новых пользователей.

    • Выход из системы.

    • Просмотр информации о текущем пользователе.

    Работа с отелями:

    • Отображение списка доступных отелей.
    • Просмотр детальной информации об отеле, включая список комнат и отзывы.

    Управление бронированиями:

    • Создание бронирований с проверкой доступности дат.
    • Отображение списка бронирований текущего пользователя.
    • Редактирование и удаление бронирований с проверкой на права доступа.

    Работа с отзывами:

    • Возможность оставлять отзывы на бронирования.
    • Редактирование уже существующих отзывов.

    Особенности реализации:

    • Использование классов-представлений: Для упрощения кода и управления функционалом в части бронирования и работы с отзывами применялись классы DetailView и UpdateView.
    • Проверка прав доступа: Все представления, связанные с бронированием и отзывами, защищены декораторами или миксинами, чтобы исключить доступ к данным других пользователей.
    • Проверка корректности данных: При создании и редактировании бронирований и отзывов введены проверки:
    • Соответствие дат заезда и выезда.
    • Наличие доступных дат для бронирования.
    • Проверка прав на изменение или удаление данных.
